cmake_minimum_required(VERSION 3.20)
project(md3_gtk_catalog C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4)
add_executable(md3-gtk-catalog main.c)
target_link_libraries(md3-gtk-catalog PRIVATE PkgConfig::GTK4)
enable_testing()
add_test(NAME smoke COMMAND md3-gtk-catalog --smoke)
